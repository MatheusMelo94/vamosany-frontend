const isDesktop = (page: Page) => (page.viewportSize()?.width ?? 0) >= 1024;
const header = (page: Page) => page.locator("header[data-nav]");
const megaPanel = (page: Page) =>
  header(page).locator(".nav-capsule div.rounded-3xl");
const field = (page: Page, label: string) =>
  page.locator("main").getByLabel(label, { exact: true });

async function hydratedButton(page: Page, name: string) {
  await expect(page.getByRole("button", { name })).toHaveAttribute(
    "type",
    "submit",
  );
}

async function axeSeriousInMain(page: Page): Promise<string[]> {
  await waitForMotionToSettle(page);
  const results = await new AxeBuilder({ page }).analyze();
  const out: string[] = [];
  for (const v of results.violations) {
    if (v.impact !== "serious" && v.impact !== "critical") continue;
    for (const n of v.nodes) {
      const target = String(n.target[0]);
      const inFooter = await page.evaluate(
        (sel) => Boolean(document.querySelector(sel)?.closest("footer")),
        target,
      );
      if (v.id === "color-contrast" && inFooter) continue; // known, ADR-0006
      out.push(`${v.id}: ${target}`);
    }
  }
  return out;
}

    page,
  }) => {
    test.skip(!isDesktop(page), "mega-menu at >= 1024 only");
    await page.goto(process.env.GAP_PATH ?? "/sobre-mi");
    await page.waitForTimeout(1200);
    const results: string[] = [];
    for (const label of (process.env.GAP_LABELS ?? "Itineraries,Sobre mí").split(",")) {
      const trigger = header(page).getByRole("button", { name: label, exact: true });
      const tb = (await trigger.boundingBox())!;
      const cap = (await header(page).locator(".nav-capsule").boundingBox())!;
      await page.mouse.move(tb.x + tb.width / 2, tb.y + tb.height / 2);
      await expect(trigger).toHaveAttribute("aria-expanded", "true");
      const pb = (await megaPanel(page).boundingBox())!;
      const gap = pb.y - (cap.y + cap.height);
      // Straight down, 1px per mousemove: a real slow hand.
      const fromY = tb.y + tb.height / 2;
      const toY = pb.y + 40;
      const steps = Math.round(toY - fromY);
      await page.mouse.move(tb.x + tb.width / 2, toY, { steps });
      const stillOpen = await trigger.getAttribute("aria-expanded");
      results.push(`${label}: gap=${gap.toFixed(1)}px steps=${steps} open=${stillOpen}`);
      // Reset
      await page.mouse.move(5, 700);
      await page.waitForTimeout(200);
    }
    test.info().annotations.push({ type: "gap-probe", description: results.join(" | ") });
    console.log("GAP PROBE", page.url(), results.join(" | "));
    expect(results.every((r) => r.endsWith("open=true")), results.join(" | ")).toBe(true);
  });

async function overflowReport(page: Page): Promise<string[]> {
  return page.evaluate(() => {
    const vw = document.documentElement.clientWidth;
    const out: string[] = [];
    const sw = document.scrollingElement!.scrollWidth;
    if (sw > vw) out.push(`scrollWidth ${sw} > ${vw}`);
    for (const el of document.querySelectorAll<HTMLElement>("main *, header *")) {
      const r = el.getBoundingClientRect();
      if (r.width === 0 || r.height === 0) continue;
      const s = getComputedStyle(el);
      if (s.visibility === "hidden" || s.opacity === "0") continue;
      if (el.closest("[aria-hidden='true']")) continue;
      if (r.right > vw + 0.5 || r.left < -0.5) {
        // ignore descendants of an element that clips horizontally
        let p: HTMLElement | null = el.parentElement;
        let clipped = false;
        while (p && p !== document.body) {
          const ps = getComputedStyle(p);
          if (ps.overflowX !== "visible" || ps.overflow === "hidden" || ps.clipPath !== "none") {
            const pr = p.getBoundingClientRect();
            if (pr.right <= vw + 0.5 && pr.left >= -0.5) {
              clipped = true;
              break;
            }
          }
          p = p.parentElement;
        }
        if (!clipped)
          out.push(
            `${el.tagName.toLowerCase()}.${(el.className?.toString() ?? "").slice(0, 40)} "${(el.textContent ?? "").trim().slice(0, 30)}" [${Math.round(r.left)},${Math.round(r.right)}]`,
          );
      }
    }
    return out.slice(0, 15);
  });
}

async function scrollThrough(page: Page) {
  await page.evaluate(async () => {
    for (let y = 0; y < document.body.scrollHeight; y += 400) {
      window.scrollTo(0, y);
      await new Promise((r) => setTimeout(r, 60));
    }
    window.scrollTo(0, document.body.scrollHeight);
    await new Promise((r) => setTimeout(r, 400));
  });
}

import { test, expect } from "@playwright/test";
test("gap threshold @qa-sobre-mi", async ({ page }) => {
  test.skip((page.viewportSize()?.width ?? 0) < 1024);
  await page.goto("/sobre-mi");
  await page.waitForTimeout(1200);
  const hdr = page.locator("header[data-nav]");
  const out: string[] = [];
  for (const px of [2, 4, 6, 8, 10, 12, 16, 24]) {
    let closed = 0;
    for (let rep = 0; rep < 4; rep++) {
      const trigger = hdr.getByRole("button", { name: "Sobre mí", exact: true });
      const tb = (await trigger.boundingBox())!;
      // start at varied offsets inside the trigger so sample phase varies
      const y0 = tb.y + tb.height / 2 + rep * 1.7;
      await page.mouse.move(tb.x + tb.width / 2, y0);
      await expect(trigger).toHaveAttribute("aria-expanded", "true");
      const pb = (await hdr.locator(".nav-capsule div.rounded-3xl").boundingBox())!;
      const toY = pb.y + 40;
      const steps = Math.max(1, Math.round((toY - y0) / px));
      await page.mouse.move(tb.x + tb.width / 2, toY, { steps });
      if ((await trigger.getAttribute("aria-expanded")) !== "true") closed++;
      await page.mouse.move(5, 700);
      await page.waitForTimeout(150);
    }
    out.push(`${px}px/sample: closed ${closed}/4`);
  }
  console.log("THRESHOLD", out.join(" | "));
});

test("tab past panel into main: panel still open over focus? @qa-sobre-mi", async ({ page }) => {
  await page.goto(process.env.GAP_PATH ?? "/sobre-mi");
  await page.waitForTimeout(1000);
  const t = page.locator("header[data-nav]").getByRole("button", { name: process.env.LBL ?? "Sobre mí", exact: true });
  await t.focus();
  await page.keyboard.press("Enter");
  const log: string[] = [];
  for (let i = 0; i < 14; i++) {
    await page.keyboard.press("Tab");
    const info = await page.evaluate(() => {
      const f = document.activeElement as HTMLElement;
      const panel = document.querySelector("header[data-nav] .nav-capsule div.rounded-3xl");
      const fr = f.getBoundingClientRect();
      let covered = false;
      if (panel) {
        const pr = panel.getBoundingClientRect();
        covered = !f.closest("header") && fr.top < pr.bottom && fr.bottom > pr.top && fr.left < pr.right && fr.right > pr.left;
      }
      return `${f.tagName}:${(f.textContent ?? "").trim().slice(0, 18)} open=${!!panel} covered=${covered}`;
    });
    log.push(info);
  }
  console.log("TABPAST", log.join(" | "));
});
