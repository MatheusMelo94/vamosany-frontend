/**
 * QA evidence spec, mobile-navigation "Índice" test pass v1 (2026-09-27).
 * Probes what the Engineer's mobile-nav.spec.ts doesn't: fonts across every
 * panel text node, tap targets on every visible control at more viewports,
 * overflow at 320 / landscape / 200% text, keyboard with a group open,
 * focus-not-obscured, rapid open/close, real-time motion sampling (settle
 * and frame pacing), and layout shift during the accordion.
 */

const header = (page: Page) => page.locator("header[data-nav]");
const burger = (page: Page) =>
  header(page).getByRole("button", { name: /^(Abrir|Cerrar) menú$/ });
const panel = (page: Page) =>
  header(page).getByRole("navigation", { name: "Menú" });
const GROUPS = ["Itineraries", "Guides", "Blog", "Services", "Sobre mí"];
const isDesktop = (page: Page) => (page.viewportSize()?.width ?? 0) >= 1024;

async function land(page: Page, path = "/") {
  await page.goto(path);
  await page.evaluate(() => document.fonts.ready);
  if (path === "/") await waitForLandingMotion(page);
  else await waitForGsapIdle(page);
}

async function open(page: Page): Promise<Locator> {
  await burger(page).click();
  await expect(panel(page)).toBeVisible();
  await waitForGsapIdle(page);
  return panel(page);
}

async function openGroup(page: Page, label: string) {
  const h = panel(page).getByRole("button", { name: label, exact: true });
  if ((await h.getAttribute("aria-expanded")) !== "true") await h.click();
  await waitForGsapIdle(page);
}

/* Fonts (spec v1.1 § Typography; AC 6, 10, 17)                        */
/* ------------------------------------------------------------------ */

type TextNode = {
  text: string;
  family: string;
  weight: string;
  transform: string;
  role: string;
};

function panelText(page: Page): Promise<TextNode[]> {
  return panel(page).evaluate((nav) => {
    const out: TextNode[] = [];
    const walker = document.createTreeWalker(nav, NodeFilter.SHOW_TEXT);
    for (let n = walker.nextNode(); n; n = walker.nextNode()) {
      const text = n.textContent?.trim() ?? "";
      const el = n.parentElement;
      if (!text || !el || el.closest("[hidden]")) continue;
      const cs = getComputedStyle(el);
      const role = el.closest("h3")
        ? "column"
        : el.closest("[data-row-price]")
          ? "price"
          : el.id.endsWith("-scent")
            ? "scent"
            : el.closest("[data-row-tag]")
              ? "tag"
              : el.closest("[data-row-title]")
                ? "title"
                : el.closest("h2")
                  ? "label"
                  : el.closest("[role=group]")
                    ? "segment"
                    : el.closest("a")
                      ? "pill"
                      : "other";
      out.push({
        text,
        family: cs.fontFamily,
        weight: cs.fontWeight,
        transform: cs.textTransform,
        role,
      });
    }
    return out;
  });
}

test.describe("QA fonts match the desktop menu (v1.1)", () => {
  test.beforeEach(({ page }) => test.skip(isDesktop(page), "drawer < 1024"));

  for (const path of ["/", "/sobre-mi"]) {
    test(`${path}: Courier only for column headings, scent lines and prices; Montserrat elsewhere`, async ({
      page,
    }) => {
      await land(page, path);
      await open(page);
      const loaded = await page.evaluate(() => ({
        mont: document.fonts.check("700 18px Montserrat"),
        courier: [...document.fonts].some(
          (f) => /courier/i.test(f.family) && f.status === "loaded",
        ),
        montFaces: [...document.fonts]
          .filter((f) => /montserrat/i.test(f.family))
          .map((f) => `${f.weight}:${f.status}`),
      }));
      expect(loaded.courier, JSON.stringify(loaded)).toBe(true);
      const seen: TextNode[] = [];
      for (const g of GROUPS) {
        await openGroup(page, g);
        seen.push(...(await panelText(page)));
      }
      const bad: string[] = [];
      for (const t of seen) {
        const courier = /courier/i.test(t.family);
        const shouldCourier = ["column", "scent", "price"].includes(t.role);
        if (courier !== shouldCourier)
          bad.push(`${t.role} "${t.text}" → ${t.family}`);
        if (!courier && !/^"?montserrat/i.test(t.family))
          bad.push(`${t.role} "${t.text}" not Montserrat-first: ${t.family}`);
        if (!courier && t.transform !== "none")
          bad.push(`${t.role} "${t.text}" text-transform ${t.transform}`);
        const w = Number(t.weight);
        const want: Record<string, number> = {
          label: 700,
          title: 600,
          tag: 600,
          segment: 600,
          pill: 600,
          price: 700,
        };
        if (want[t.role] && w !== want[t.role])
          bad.push(`${t.role} "${t.text}" weight ${w}`);
      }
      expect(seen.filter((t) => t.role === "other")).toEqual([]);
      expect(bad).toEqual([]);
    });
  }
});

const VIEWPORTS: Array<[number, number]> = [
  [360, 780],
  [390, 844],
  [360, 640],
  [320, 780],
  [768, 1024],
  [780, 360],
  [640, 360],
];

function targets(page: Page) {
  return panel(page).evaluate((nav) => {
    const list = nav.querySelector("[data-drawer-list]")!.getBoundingClientRect();
    return [...nav.querySelectorAll<HTMLElement>("a[href], button")]
      .filter((el) => !el.closest("[hidden]"))
      .map((el) => {
        const r = el.getBoundingClientRect();
        const kind = el.closest("h2")
          ? "header"
          : el.closest("[role=group]")
            ? "segment"
            : el.closest("[data-drawer-footer]")
              ? "pill"
              : "row";
        return {
          kind,
          name: (el.textContent ?? "").trim().slice(0, 30),
          h: Math.round(r.height * 10) / 10,
          w: Math.round(r.width * 10) / 10,
          listW: Math.round(list.width - 40),
        };
      });
  });
}

test.describe("QA tap targets at every viewport of record", () => {
  test.skip(({ browserName }) => browserName !== "chromium");
  for (const [w, h] of VIEWPORTS) {
    test(`${w} × ${h}: rows ≥44 full width, headers ≥56, pills 48, segments ≥44 × 44`, async ({
      page,
    }) => {
      await page.setViewportSize({ width: w, height: h });
      await land(page);
      await open(page);
      const bad: string[] = [];
      for (const g of GROUPS) {
        await openGroup(page, g);
        for (const t of await targets(page)) {
          const min = { header: 56, segment: 44, pill: 48, row: 44 }[t.kind]!;
          if (t.h < min - 0.5) bad.push(`${g}/${t.kind} "${t.name}" h=${t.h}`);
          if (t.kind === "segment" && t.w < 44)
            bad.push(`${t.kind} "${t.name}" w=${t.w}`);
          if (t.kind === "row" && Math.abs(t.w - t.listW) > 1)
            bad.push(`row "${t.name}" w=${t.w} vs list ${t.listW}`);
        }
      }
      const b = await burger(page).boundingBox();
      test.info().annotations.push({
        type: "burger",
        description: `${b?.width}×${b?.height}`,
      });
      expect(bad).toEqual([]);
    });
  }
});

/* Overflow: 320, landscape, 639/640, 200% text (AC 50, 51; § A11y)     */
/* ------------------------------------------------------------------ */

function overflow(page: Page) {
  return page.evaluate(() => {
    const nav = document.querySelector<HTMLElement>(
      "header[data-nav] nav[aria-label='Menú']",
    )!;
    const pr = nav.getBoundingClientRect();
    const out: string[] = [];
    if (document.documentElement.scrollWidth > innerWidth)
      out.push(`doc ${document.documentElement.scrollWidth} > ${innerWidth}`);
    for (const el of nav.querySelectorAll<HTMLElement>("*")) {
      if (el.closest("[hidden]")) continue;
      if (el.scrollWidth > el.clientWidth + 1 && getComputedStyle(el).overflowX !== "visible")
        out.push(`scroll-x ${el.tagName}.${el.className.toString().slice(0, 40)}`);
      const r = el.getBoundingClientRect();
      if (r.width && (r.right > pr.right + 0.5 || r.left < pr.left - 0.5))
        out.push(`${el.tagName} "${(el.textContent ?? "").trim().slice(0, 24)}" ${r.left.toFixed(1)}–${r.right.toFixed(1)} outside ${pr.left}–${pr.right}`);
      // Truncation: text clipped with ellipsis or hidden overflow on a text leaf.
      const cs = getComputedStyle(el);
      if (cs.textOverflow === "ellipsis") out.push(`ellipsis ${el.tagName}`);
    }
    return out;
  });
}

async function drawerState(page: Page) {
  return page.evaluate(() => {
    const nav = document.querySelector<HTMLElement>("header[data-nav] nav[aria-label='Menú']");
    return {
      open: !!nav,
      clip: nav?.style.clipPath ?? "",
      inert: document.querySelectorAll("[inert]").length,
      overflow: document.body.style.overflow,
      op: nav
        ? [...nav.querySelectorAll<HTMLElement>("[data-drawer-group],[data-drawer-footer]")].map(
            (e) => getComputedStyle(e).opacity,
          )
        : [],
      burger: document
        .querySelector("header[data-nav] button[aria-label$='menú']")
        ?.getAttribute("aria-label"),
      iconStyle: document.querySelector("header[data-nav] button[aria-label$='menú'] svg")?.getAttribute("style") ?? "",
    };
  });
}

test.describe("QA rapid taps", () => {
  test.beforeEach(({ page }) => test.skip(isDesktop(page), "drawer < 1024"));
  for (const gap of [80, 200, 400]) {
    test(`7 taps ${gap}ms apart end open and fully settled; 8 end closed and clean`, async ({ page }) => {
      await land(page);
      for (let i = 0; i < 7; i++) {
        await burger(page).click({ delay: 0 });
        await page.waitForTimeout(gap);
      }
      await waitForGsapIdle(page);
      await page.waitForTimeout(200);
      const s = await drawerState(page);
      expect(s.open).toBe(true);
      expect(s.clip).toBe("");
      expect(s.op.every((o) => o === "1")).toBe(true);
      expect(s.overflow).toBe("hidden");
      expect(s.burger).toBe("Cerrar menú");
      await burger(page).click();
      await waitForGsapIdle(page);
      await expect(panel(page)).toHaveCount(0);
      const c = await drawerState(page);
      expect(c).toMatchObject({ open: false, inert: 0, overflow: "", burger: "Abrir menú" });
      expect(c.iconStyle).toBe("");
    });
  }

  test("open a group, close mid-accordion, reopen at once: the default state, list at top, nothing stuck", async ({
    page,
  }) => {
    await land(page, "/contacto");
    await open(page);
    await openGroup(page, "Itineraries");
    await panel(page).getByRole("button", { name: "Guides", exact: true }).click();
    await page.waitForTimeout(60);
    await burger(page).click();
    await page.waitForTimeout(100);
    await burger(page).click();
    await waitForGsapIdle(page);
    const s = await panel(page).evaluate((nav) => ({
      expanded: [...nav.querySelectorAll("h2 button")].map((b) => b.getAttribute("aria-expanded")),
      shown: [...nav.querySelectorAll("[data-drawer-body]")].map((b) => !b.hasAttribute("hidden")),
      heights: [...nav.querySelectorAll<HTMLElement>("[data-drawer-body]")].map((b) => b.style.height),
      top: nav.querySelector("[data-drawer-list]")!.scrollTop,
    }));
    expect(s.expanded).toEqual(["false", "false", "false", "false", "true"]);
    expect(s.shown).toEqual([false, false, false, false, true]);
    expect(s.heights.every((h) => h === "")).toBe(true);
    expect(s.top).toBe(0);
  });
});

test.describe("QA motion, real time", () => {
  test.beforeEach(({ page, browserName }) =>
    test.skip(isDesktop(page) || browserName !== "chromium", "drawer < 1024"),
  );

  for (const throttle of [1, 4]) {
    test(`unfold timeline and frame pacing, CPU ×${throttle}`, async ({ page }) => {
      test.setTimeout(120_000);
      await land(page);
      // Warm the lazy chunk so the first open isn't a network measurement.
      await burger(page).click();
      await waitForGsapIdle(page);
      await burger(page).click();
      await waitForGsapIdle(page);
      await expect(panel(page)).toHaveCount(0);
      const cdp = await page.context().newCDPSession(page);
      await cdp.send("Emulation.setCPUThrottlingRate", { rate: throttle });
      const runs: Array<Record<string, number>> = [];
      for (let run = 0; run < 3; run++) {
        const sampler = await sampleFrames(
          page,
          () => {
            const nav = document.querySelector<HTMLElement>("header[data-nav] nav[aria-label='Menú']");
            const veil = nav?.previousElementSibling as HTMLElement | null;
            const op = (e: Element | null) => (e ? Number(getComputedStyle(e).opacity) : 0);
            return {
              t: performance.now(),
              clip: nav ? getComputedStyle(nav).clipPath : "absent",
              veil: op(veil),
              groups: nav ? [...nav.querySelectorAll("[data-drawer-group]")].map(op) : [],
              footer: op(nav?.querySelector("[data-drawer-footer]") ?? null),
            };
          },
          null,
        );
        const t0 = await page.evaluate(() => performance.now());
        await burger(page).click();
        await page.waitForTimeout(1500 * throttle);
        const s = (await sampler.stop()) as Sample[];
        const after = s.filter((x) => x.t >= t0);
        const first = (p: (x: Sample) => boolean) => {
          const hit = after.find(p);
          return hit ? Math.round(hit.t - t0) : -1;
        };
        const gaps = after.slice(1).map((x, i) => x.t - after[i]!.t);
        const windowed = after.filter((x) => x.t - t0 < 900);
        const wgaps = windowed.slice(1).map((x, i) => x.t - windowed[i]!.t);
        runs.push({
          firstPaint: first((x) => x.clip !== "absent"),
          panelUnclipped: first((x) => x.clip === "none"),
          lastHeaderOpaque: first((x) => x.groups.length === 5 && x.groups.every((o) => o >= 0.99)),
          allHeadersReadable: first((x) => x.groups.length === 5 && x.groups.every((o) => o >= 0.8)),
          footerOpaque: first((x) => x.footer >= 0.99),
          veilOpaque: first((x) => x.veil >= 0.99),
          framesIn900: wgaps.length,
          maxFrameGap: Math.round(Math.max(...wgaps, 0)),
          longFrames: wgaps.filter((g) => g > 34).length,
          medianGap: Math.round([...gaps].sort((a, b) => a - b)[Math.floor(gaps.length / 2)] ?? 0),
        });
        await burger(page).click();
        await page.waitForTimeout(600 * throttle);
        await expect(panel(page)).toHaveCount(0);
      }
      await cdp.send("Emulation.setCPUThrottlingRate", { rate: 1 });
      test.info().annotations.push({ type: `unfold ×${throttle}`, description: JSON.stringify(runs) });
      // Settles within the spec's 0.9s envelope (+1 frame) at ×1.
      if (throttle === 1) for (const r of runs) expect(r.lastHeaderOpaque).toBeLessThan(950);
    });
  }

  test("tap-to-next-paint latency on the burger (INP proxy), ×1 and ×4", async ({ page }) => {

    await land(page);
    await burger(page).click();
    await waitForGsapIdle(page);
    await burger(page).click();
    await waitForGsapIdle(page);
    const cdp = await page.context().newCDPSession(page);
    const out: Record<string, number[]> = {};
    for (const rate of [1, 4]) {
      await cdp.send("Emulation.setCPUThrottlingRate", { rate });
      await page.evaluate(() => {
        (window as unknown as { __inp: number[] }).__inp = [];
        new PerformanceObserver((l) => {
          for (const e of l.getEntries())
            (window as unknown as { __inp: number[] }).__inp.push(Math.round(e.duration));
        }).observe({ type: "event", durationThreshold: 16, buffered: false } as PerformanceObserverInit);
      });
      for (let i = 0; i < 6; i++) {
        await burger(page).tap().catch(() => burger(page).click());
        await page.waitForTimeout(1200 * rate);
      }
      out[`x${rate}`] = await page.evaluate(() => (window as unknown as { __inp: number[] }).__inp);
    }
    await cdp.send("Emulation.setCPUThrottlingRate", { rate: 1 });
    test.info().annotations.push({ type: "event durations (ms)", description: JSON.stringify(out) });
  });

  test("accordion: layout-shift entries and header stability during expand/collapse", async ({ page }) => {
    await land(page);
    await open(page);
    await page.evaluate(() => {
      const w = window as unknown as { __ls: Array<{ v: number; input: boolean; srcs: string[] }> };
      w.__ls = [];
      new PerformanceObserver((l) => {
        for (const e of l.getEntries() as Array<PerformanceEntry & { value: number; hadRecentInput: boolean; sources?: Array<{ node?: Node }> }>)
          w.__ls.push({
            v: e.value,
            input: e.hadRecentInput,
            srcs: (e.sources ?? []).map((s) => (s.node as Element | undefined)?.textContent?.trim().slice(0, 20) ?? "?"),
          });
      }).observe({ type: "layout-shift", buffered: false });
    });
    // Sample the tapped header's top and the capsule/main during each tap.
    const tops: Record<string, number[]> = {};
    for (const g of ["Guides", "Blog", "Guides", "Itineraries", "Sobre mí", "Sobre mí"]) {
      const sampler = await sampleFrames(
        page,
        (label: string) => {
          const b = [...document.querySelectorAll("header[data-nav] h2 button")].find(
            (x) => x.textContent?.includes(label),
          )!;
          return {
            top: Math.round(b.getBoundingClientRect().top * 10) / 10,
            cap: document.querySelector(".nav-capsule")?.getBoundingClientRect().top ?? 0,
          };
        },
        g,
      );
      await panel(page).getByRole("button", { name: g, exact: true }).click();
      await page.waitForTimeout(700);
      const s = (await sampler.stop()) as Array<{ top: number; cap: number }>;
      tops[`${g}#${Object.keys(tops).length}`] = [...new Set(s.map((x) => x.top))];
      expect(new Set(s.map((x) => x.cap)).size).toBe(1);
    }
    const ls = await page.evaluate(() => (window as unknown as { __ls: unknown[] }).__ls);
    test.info().annotations.push({ type: "layout-shift", description: JSON.stringify(ls) });
    test.info().annotations.push({ type: "tapped header tops", description: JSON.stringify(tops) });
    const unexpected = (ls as Array<{ v: number; input: boolean }>).filter((e) => !e.input);
    const cls = unexpected.reduce((a, e) => a + e.v, 0);
    expect(cls).toBeLessThan(0.1);
  });
